// Archivo: scripts/i5/04-rol.ts
// Cambia el rol de un usuario. Es la ÚNICA forma de crear un administrador: nunca desde la app
// Uso: node --env-file=.env scripts/i5/04-rol.ts bea@entrenos.test admin
import { MongoClient } from "mongodb";

const [email, rol] = process.argv.slice(2);
if (!email || (rol !== "admin" && rol !== "user")) {
  console.error("Uso: node --env-file=.env scripts/i5/04-rol.ts <email> admin|user");
  process.exit(1);
}
const cliente = await MongoClient.connect(process.env.MONGODB_URI!);
const r = await cliente.db(process.env.MONGODB_DB).collection("user").updateOne({ email }, { $set: { rol } });
console.log(r.matchedCount ? `${email} ahora es ${rol}` : `No existe ningún usuario ${email}`);
await cliente.close();
